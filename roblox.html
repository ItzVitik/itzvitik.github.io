local ReplicatedStorage = game:GetService("ReplicatedStorage")local Players = game:GetService("Players") local Items = require(ReplicatedStorage.Shared.Items)local Purchase = ReplicatedStorage.Remotes.Purchase local COOLDOWN = 0.5local lastBuy = {} Purchase.OnServerInvoke = function(player, itemId)    if typeof(itemId) ~= "string" then return false, "Bad request" end    local item = Items[itemId]    if not item then return false, "Unknown item" end     local now = os.clock()    if (lastBuy[player] or 0) + COOLDOWN > now then return false, "Slow down" end    lastBuy[player] = now     local cash = player.leaderstats.Cash    if cash.Value < item.price then return false, "Not enough cash" end     cash.Value -= item.price    item.grant(player)    return trueend Players.PlayerRemoving:Connect(function(player)    lastBuy[player] = nilend)
