Server decides, client asks.

A trimmed example of how I wire a shop purchase. The client only sends a request, the server checks everything before money moves.

ServerScriptService / ShopService.server.lua 
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local Players = game:GetService("Players")

local Items = require(ReplicatedStorage.Shared.Items)
local Purchase = ReplicatedStorage.Remotes.Purchase

local COOLDOWN = 0.5
local lastBuy = {}

Purchase.OnServerInvoke = function(player, itemId)
    -- Never trust what the client sends
    if typeof(itemId) ~= "string" then return false, "Bad request" end
    local item = Items[itemId]
    if not item then return false, "Unknown item" end

    local now = os.clock()
    if (lastBuy[player] or 0) + COOLDOWN > now then return false, "Slow down" end
    lastBuy[player] = now

    local cash = player.leaderstats.Cash
    if cash.Value < item.price then return false, "Not enough cash" end

    cash.Value -= item.price
    item.grant(player)
    return true
end

Players.PlayerRemoving:Connect(function(player)
    lastBuy[player] = nil
end)
